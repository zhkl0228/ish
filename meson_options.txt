# Gone with the x86 guest (fb0351e2): the guest is AArch64 only. Kept, with
# arm64 its only value, because a build dir set up with -Dguest_arch=arm64
# (every one of the fork's: the default was x86) has it among its user
# options, and meson 1.11 refuses to regenerate one with an option the
# project no longer has ("Unknown options"). Setting it says it is
# deprecated; x86 is refused.
option('guest_arch', type: 'combo', choices: ['arm64'], value: 'arm64', deprecated: true,
       description: 'Deprecated and ignored: the guest is AArch64 only')

# Per-offload verification handlers (dev only, ARM64). Each offload type has
# its own independent switch + test files, so they compile in isolation.

       description: 'Build symbol-level offload test handlers (mem*/hash/dict).')
option('offload_test_prebuilt', type: 'boolean', value: false,
       description: 'Build prebuilt-gadget offload test handlers (spec_mix).')

# dual mapping on iOS). Off by default; ISH_JIT=0 disables it at run time.

       description: 'Build the ARM64 guest with the native JIT.')
option('jit_emit', type: 'boolean', value: true,
       description: 'With jit: translate at run time (needs executable memory). false runs only AOT images linked into the target.')
option('cli_aot', type: 'array', value: [],
       description: 'AOT images (aot_<name>.S from tools/jit_aot/gen.py, or their .o) to link into the ish CLI executable only; needs jit.')
